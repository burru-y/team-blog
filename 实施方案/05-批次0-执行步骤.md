# 批次 0 · 地基：逐步执行

> 目标：一条命令拉起数据库，空库能迁移到最新，`/api/v1/health` 返回 200，种子数据能让热力图查询出数。
> 时长：0.5–1 天（单人）。**这一步做错，后面 11 天的返工量是它的 10 倍。**

## 步骤 0.2 · 生成 Spring 骨架（15 分钟）

不要手写 pom。用官方生成器，依赖一次点对：

```bash
curl -G https://start.spring.io/starter.zip \
  -d type=maven-project \
  -d language=java \
  -d bootVersion=3.4.1 \
  -d groupId=com.teamblog \
  -d artifactId=api \
  -d name=api \
  -d packageName=com.teamblog \
  -d javaVersion=21 \
  -d dependencies=web,validation,data-jpa,security,flyway,postgresql,actuator,configuration-processor,lombok \
  -o api.zip

unzip -q api.zip -d api-tmp && mv api-tmp/api/* api/ && rm -rf api-tmp api.zip
```

`bootVersion` 以站点当前 3.4.x 为准（`curl -s https://start.spring.io/metadata/client | jq '.bootVersion'`）。

**检查 `api/pom.xml` 里必须有这两个依赖**（Boot 3.3 起 Flyway 拆了数据库模块，只有 `flyway-core` 会在启动时报 `No database found to handle jdbc:postgresql`）：

```xml
<dependency>
  <groupId>org.flywaydb</groupId>
  <artifactId>flyway-core</artifactId>
</dependency>
<dependency>
  <groupId>org.flywaydb</groupId>
  <artifactId>flyway-database-postgresql</artifactId>
</dependency>
```

生成器没加第二条时手动补，版本由 Boot 的 BOM 管理，不写 `<version>`。

`ops/init/01-roles.sql`（容器**首次**初始化时以超级用户执行，之后再加用户要手动跑）：

```sql
-- AI 服务专用只读账号：物理上不能改业务表
DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'ai_reader') THEN
    CREATE ROLE ai_reader LOGIN PASSWORD 'ai_dev_pw';
  END IF;
END $$;

GRANT CONNECT ON DATABASE blog TO ai_reader;
GRANT USAGE ON SCHEMA public TO ai_reader;

-- 关键：默认权限要指定 FOR ROLE，因为表是 blog 建的
ALTER DEFAULT PRIVILEGES FOR ROLE blog IN SCHEMA public
  GRANT SELECT ON TABLES TO ai_reader;
```

启动并确认时区（**这条一定要查**，UTC 会让打卡日期错一天）：

```bash
docker compose -f ops/docker-compose.dev.yml up -d
docker compose -f ops/docker-compose.dev.yml ps          # db 应为 healthy
psql "postgresql://blog:blog_dev_pw@localhost:5432/blog" -c "show timezone;"
# 期望：Asia/Shanghai
```

## 步骤 0.4 · Flyway V1（2 小时，核心）

`api/src/main/resources/db/migration/V1__init.sql`

> 说明：主键由应用侧生成 UUIDv7（`java.util.UUID`），所以这里只声明 `uuid` 类型，不设默认值；种子数据例外，用 `gen_random_uuid()`。

```sql
-- ============ 扩展 ============
-- pgcrypto 只用于种子里生成 bcrypt 哈希；PG13+ 的 gen_random_uuid 是内置的
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- ============ 用户 ============
CREATE TABLE users (
    id            uuid PRIMARY KEY,
    username      varchar(20)  NOT NULL,
    email         varchar(255) NOT NULL,
    password_hash varchar(100) NOT NULL,
    role          varchar(10)  NOT NULL DEFAULT 'MEMBER',
    avatar_url    varchar(500),
    bio           text,
    github_url    varchar(300),
    email_public  boolean      NOT NULL DEFAULT false,
    total_points  integer      NOT NULL DEFAULT 0,
    token_version integer      NOT NULL DEFAULT 0,
    last_login_at timestamptz,
    deleted_at    timestamptz,
    created_at    timestamptz  NOT NULL DEFAULT now(),
    CONSTRAINT uq_users_username UNIQUE (username),
    CONSTRAINT uq_users_email    UNIQUE (email),
    CONSTRAINT ck_users_role     CHECK (role IN ('ADMIN', 'MEMBER')),
    CONSTRAINT ck_users_username CHECK (username ~ '^[a-z0-9_]{3,20}$'),
    CONSTRAINT ck_users_points   CHECK (total_points >= 0)
);

-- 邮箱大小写不敏感（注册时也统一转小写，双保险）
CREATE UNIQUE INDEX uq_users_email_lower ON users (lower(email));

CREATE TABLE refresh_tokens (
    id             uuid PRIMARY KEY,
    user_id        uuid        NOT NULL REFERENCES users (id) ON DELETE CASCADE,
    token_hash     varchar(64) NOT NULL,
    expires_at     timestamptz NOT NULL,
    revoked_at     timestamptz,
    replaced_by_id uuid,
    ip_hash        varchar(64),
    created_at     timestamptz NOT NULL DEFAULT now(),
    CONSTRAINT uq_refresh_hash UNIQUE (token_hash)
);
CREATE INDEX ix_refresh_user_exp ON refresh_tokens (user_id, expires_at);
CREATE INDEX ix_refresh_expires  ON refresh_tokens (expires_at);

-- ============ 成果 ============
CREATE TABLE projects (
    id          uuid PRIMARY KEY,
    owner_id    uuid         NOT NULL REFERENCES users (id),
    type        varchar(20)  NOT NULL,
    status      varchar(10)  NOT NULL DEFAULT 'ACTIVE',
    title       varchar(200) NOT NULL,
    description text         NOT NULL,
    github_url  varchar(300),
    content_md  text,
    image_url   varchar(500),
    removed_at  timestamptz,
    created_at  timestamptz  NOT NULL DEFAULT now(),
    updated_at  timestamptz  NOT NULL DEFAULT now(),
    CONSTRAINT ck_projects_type   CHECK (type IN ('PROJECT', 'CERTIFICATE', 'AWARD')),
    CONSTRAINT ck_projects_status CHECK (status IN ('ACTIVE', 'REMOVED')),
    -- v1.0 必填字段在数据库层兜底：项目必须有链接+内容，证书/奖项必须有图或链接
    CONSTRAINT ck_projects_required CHECK (
        (type = 'PROJECT' AND github_url IS NOT NULL AND content_md IS NOT NULL)
        OR (type IN ('CERTIFICATE', 'AWARD') AND (image_url IS NOT NULL OR github_url IS NOT NULL))
    ),
    CONSTRAINT ck_projects_title CHECK (char_length(btrim(title)) > 0)
);
CREATE INDEX ix_projects_feed  ON projects (status, created_at DESC);
CREATE INDEX ix_projects_owner ON projects (owner_id, status);
CREATE INDEX ix_projects_type  ON projects (type, status);

-- ============ 学习内容 ============
CREATE TABLE learning_logs (
    id         uuid PRIMARY KEY,
    user_id    uuid         NOT NULL REFERENCES users (id),
    title      varchar(200) NOT NULL,
    content_md text         NOT NULL,
    summary_md text,
    type       varchar(10)  NOT NULL DEFAULT 'NOTE',
    log_date   date         NOT NULL,
    created_at timestamptz  NOT NULL DEFAULT now(),
    CONSTRAINT ck_logs_type  CHECK (type IN ('NOTE', 'CODE')),
    CONSTRAINT ck_logs_title CHECK (char_length(btrim(title)) > 0)
);
CREATE INDEX ix_logs_user_created ON learning_logs (user_id, created_at DESC);
CREATE INDEX ix_logs_created      ON learning_logs (created_at DESC);
CREATE INDEX ix_logs_log_date     ON learning_logs (log_date);

-- ============ 打卡与积分 ============
CREATE TABLE checkins (
    id           uuid PRIMARY KEY,
    user_id      uuid         NOT NULL REFERENCES users (id),
    checkin_date date         NOT NULL,
    note         varchar(200),
    created_at   timestamptz  NOT NULL DEFAULT now(),
    CONSTRAINT uq_checkins_user_date UNIQUE (user_id, checkin_date)
);
CREATE INDEX ix_checkins_date ON checkins (checkin_date);

CREATE TABLE point_events (
    id         uuid PRIMARY KEY,
    user_id    uuid         NOT NULL REFERENCES users (id),
    kind       varchar(20)  NOT NULL,
    ref_id     varchar(64)  NOT NULL,
    points     integer      NOT NULL,
    created_at timestamptz  NOT NULL DEFAULT now(),
    CONSTRAINT ck_points_kind  CHECK (kind IN ('CHECKIN', 'LEARNING_LOG', 'PROJECT',
                                               'STREAK_7', 'STREAK_14', 'STREAK_30')),
    CONSTRAINT ck_points_value CHECK (points > 0),
    -- “只加一次”的最终保证：同一事件唯一
    CONSTRAINT uq_points_once  UNIQUE (user_id, kind, ref_id)
);
CREATE INDEX ix_points_user_created ON point_events (user_id, created_at DESC);

-- ============ 评论（两级） ============
CREATE TABLE comments (
    id          uuid PRIMARY KEY,
    user_id     uuid         NOT NULL REFERENCES users (id),
    target_type varchar(20)  NOT NULL,
    target_id   uuid         NOT NULL,
    parent_id   uuid REFERENCES comments (id),
    content     text         NOT NULL,
    deleted_at  timestamptz,
    created_at  timestamptz  NOT NULL DEFAULT now(),
    CONSTRAINT ck_comments_target CHECK (target_type IN ('PROJECT', 'LEARNING_LOG')),
    CONSTRAINT ck_comments_len    CHECK (char_length(btrim(content)) BETWEEN 1 AND 2000),
    CONSTRAINT ck_comments_not_self CHECK (parent_id IS NULL OR parent_id <> id)
);
CREATE INDEX ix_comments_target ON comments (target_type, target_id, created_at);
CREATE INDEX ix_comments_parent ON comments (parent_id);

-- 数据库层禁止“回复的回复”，比只靠 service 判断可靠
CREATE OR REPLACE FUNCTION check_comment_depth() RETURNS trigger AS $$
BEGIN
    IF NEW.parent_id IS NOT NULL THEN
        IF EXISTS (SELECT 1 FROM comments WHERE id = NEW.parent_id AND parent_id IS NOT NULL) THEN
            RAISE EXCEPTION 'REPLY_DEPTH_EXCEEDED';
        END IF;
    END IF;
    RETURN NEW;
END $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_comment_depth
    BEFORE INSERT ON comments
    FOR EACH ROW EXECUTE FUNCTION check_comment_depth();

-- ============ 留言反馈 ============
CREATE TABLE feedbacks (
    id         uuid PRIMARY KEY,
    name       varchar(50)  NOT NULL,
    email      varchar(255),
    content    text         NOT NULL,
    status     varchar(10)  NOT NULL DEFAULT 'NEW',
    ip_hash    varchar(64),
    handled_by uuid REFERENCES users (id),
    created_at timestamptz  NOT NULL DEFAULT now(),
    CONSTRAINT ck_feedback_status CHECK (status IN ('NEW', 'READ', 'ARCHIVED')),
    CONSTRAINT ck_feedback_len    CHECK (char_length(btrim(content)) BETWEEN 1 AND 2000)
);
CREATE INDEX ix_feedback_status ON feedbacks (status, created_at DESC);

-- ============ 邀请 ============
CREATE TABLE admin_invites (
    id         uuid PRIMARY KEY,
    token_hash varchar(64)  NOT NULL,
    email      varchar(255),
    role       varchar(10)  NOT NULL DEFAULT 'MEMBER',
    created_by uuid         NOT NULL REFERENCES users (id),
    expires_at timestamptz  NOT NULL,
    used_at    timestamptz,
    created_at timestamptz  NOT NULL DEFAULT now(),
    CONSTRAINT uq_invite_hash UNIQUE (token_hash),
    CONSTRAINT ck_invite_role CHECK (role IN ('ADMIN', 'MEMBER'))
);
CREATE INDEX ix_invites_exp ON admin_invites (expires_at);

-- ============ AI ============
CREATE TABLE ai_conversations (
    id              uuid PRIMARY KEY,
    visitor_ip_hash varchar(64),
    project_id      uuid REFERENCES projects (id),
    messages        jsonb NOT NULL DEFAULT '[]'::jsonb,
    created_at      timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX ix_ai_conv_project ON ai_conversations (project_id, created_at DESC);
CREATE INDEX ix_ai_conv_created ON ai_conversations (created_at);

-- 全局/单 IP 日调用计数：scope = 'GLOBAL' 或 IP 哈希
CREATE TABLE ai_usage_daily (
    day   date        NOT NULL,
    scope varchar(64) NOT NULL,
    calls integer     NOT NULL DEFAULT 0,
    PRIMARY KEY (day, scope)
);

-- ============ 给 AI 服务的只读视图 ============
CREATE VIEW v_projects_public AS
SELECT id, type, title, description, content_md, github_url
FROM projects
WHERE status = 'ACTIVE';

-- ============ updated_at 兜底 ============
CREATE OR REPLACE FUNCTION set_updated_at() RETURNS trigger AS $$
BEGIN
    NEW.updated_at = now();
    RETURN NEW;
END $$ LANGUAGE plpgsql;

CREATE TRIGGER trg_projects_updated
    BEFORE UPDATE ON projects
    FOR EACH ROW EXECUTE FUNCTION set_updated_at();
```

public record ApiResponse<T>(boolean ok, T data, ErrorBody error) {
    public static <T> ApiResponse<T> ok(T data) { return new ApiResponse<>(true, data, null); }
    public static <T> ApiResponse<T> fail(String code, String message) {

`api/src/main/java/com/teamblog/health/HealthController.java`：

```java
@RestController
@RequestMapping("/api/v1/health")
public class HealthController {

    private final JdbcTemplate jdbc;
    private final Flyway flyway;

    public HealthController(JdbcTemplate jdbc, Flyway flyway) {
        this.jdbc = jdbc;
        this.flyway = flyway;
    }

    @GetMapping
    public ApiResponse<Map<String, Object>> health() {
        jdbc.queryForObject("select 1", Integer.class);   // 连不上直接抛 503
        var current = flyway.info().current();
        return ApiResponse.ok(Map.of(
                "status", "UP",
                "db", "UP",
                "flyway", current == null ? "NONE" : current.getVersion().getVersion()
        ));
    }
}
```

## 步骤 0.6 · 种子数据（1 小时）

`api/src/main/resources/db/seed/R__seed_dev.sql`（`R__` 前缀 = 可重复执行，内容变化会重跑；只在 dev profile 加载）：

```sql
-- 密码统一为 ChangeMe123!，用 pgcrypto 生成 $2a$12$ 格式，Spring 的 BCryptPasswordEncoder 可直接校验
INSERT INTO users (id, username, email, password_hash, role, bio, github_url, total_points)
SELECT gen_random_uuid(),
       'admin' || n,
       'admin' || n || '@teamblog.local',
       crypt('ChangeMe123!', gen_salt('bf', 12)),
       'ADMIN',
       '负责 ' || (ARRAY['后端与部署','前端','AI','测试与运营'])[n],
       'https://github.com/teamblog',
       0
FROM generate_series(1, 4) AS n
ON CONFLICT (username) DO NOTHING;

INSERT INTO users (id, username, email, password_hash, role, bio, total_points)
SELECT gen_random_uuid(),
       'member' || n,
       'member' || n || '@teamblog.local',
       crypt('ChangeMe123!', gen_salt('bf', 12)),
       'MEMBER',
       '演示成员 ' || n,
       0
FROM generate_series(1, 8) AS n
ON CONFLICT (username) DO NOTHING;

-- 近 60 天随机打卡（团队热力图需要数据）
INSERT INTO checkins (id, user_id, checkin_date)
SELECT gen_random_uuid(), u.id, d::date
FROM users u
CROSS JOIN generate_series(current_date - 59, current_date, interval '1 day') AS d
WHERE u.username LIKE 'member%'
  AND random() < 0.55
ON CONFLICT (user_id, checkin_date) DO NOTHING;

-- 每次打卡补一条积分流水（顺带验证账本结构）
INSERT INTO point_events (id, user_id, kind, ref_id, points)
SELECT gen_random_uuid(), c.user_id, 'CHECKIN', c.id::text, 1
FROM checkins c
ON CONFLICT (user_id, kind, ref_id) DO NOTHING;

-- 总分重算：验证“总分 = 流水之和”
UPDATE users u
SET total_points = COALESCE((SELECT sum(p.points) FROM point_events p WHERE p.user_id = u.id), 0);

-- 三类成果各一条（验证 CHECK 约束与列表接口）
INSERT INTO projects (id, owner_id, type, title, description, github_url, content_md, image_url)
SELECT gen_random_uuid(), u.id, v.type, v.title, v.description, v.github_url, v.content_md, v.image_url
FROM (SELECT id FROM users WHERE username = 'member1') u
CROSS JOIN (VALUES
    ('PROJECT',     '团队博客脚手架', '把登录、打卡、成果串起来的最小实现', 'https://github.com/teamblog/web', '## 技术栈\nNext.js + Spring Boot + FastAPI', NULL),
    ('CERTIFICATE', '蓝桥杯省一',     '2026 年省赛',                       NULL, NULL, '/uploads/demo/cert.png'),
    ('AWARD',       '校园黑客松三等奖', '48 小时开发',                      NULL, NULL, '/uploads/demo/award.png')
) AS v(type, title, description, github_url, content_md, image_url)
ON CONFLICT DO NOTHING;
```

> 种子里的密码是明文写死的，**只允许存在于 dev**。生产管理员密码走 `ADMIN_BOOTSTRAP` 环境变量 + 首次登录强制修改。

## 常见失败

| 现象 | 原因 | 处理 |
|---|---|---|
| `No database found to handle jdbc:postgresql` | 缺 `flyway-database-postgresql` | 按 0.2 补依赖 |
| `Flyway validate failed: checksum mismatch` | 改过已执行的 V1 | 本地 `migrate clean` 重建；**已上线的迁移永不修改**，加 V2 |
| `ERROR: function gen_salt(unknown, integer) does not exist` | 没装 pgcrypto | V1 开头 `CREATE EXTENSION` |
| 打卡日期差一天 | 容器/连接时区是 UTC | 查 `show timezone;`，compose 里 `-c timezone=Asia/Shanghai` |
| `permission denied for schema public`（ai_reader） | 忘了 `GRANT USAGE ON SCHEMA` | 重跑 `ops/init/01-roles.sql` 的 grant 段 |
| `relation "users" does not exist` 但迁移显示成功 | 看错了库（连到默认 `postgres` 库） | 检查连接串里的 `/blog` |
| Hibernate 报 `missing column` / `wrong column type` | 实体与 V1 不一致 | 改实体，不要改 V1；`ddl-auto=validate` 就是在替你挡这个 |
| 5432 端口被占用 | 本机装了 PostgreSQL 服务 | 停本机服务或把 compose 映射改成 5433 |
